Supabase DB 테이블 생성 & RLS 보안 정책 SQL

Supabase SQL Editor에 복사하여 붙여넣으면 즉시 테이블과 권한이 생성됩니다.

📌 적용 방법 (3단계):

  1. Supabase 대시보드 접속 후 프로젝트 선택
  2. 좌측 메뉴에서 SQL Editor 클릭
  3. 아래 SQL 스크립트를 '전체 SQL 복사' 버튼으로 복사 후 붙여넣고 Run (실행) 클릭
-- =========================================================================
-- 1. 광화문 맛집(restaurants) 테이블 생성
-- =========================================================================
create table if not exists public.restaurants (
  id text primary key,
  name text not null,
  category text not null,
  signature_menu text,
  price integer not null default 0,
  rating numeric(3, 1) default 4.5,
  likes integer default 0,
  address text,
  lat double precision,
  lng double precision,
  author text default '광화문직장인',
  reviews jsonb default '[]'::jsonb,
  user_id uuid references auth.users(id) on delete set null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- =========================================================================
-- 2. 검색 및 필터 성능 최적화 인덱스 생성
-- =========================================================================
create index if not exists idx_restaurants_category on public.restaurants (category);
create index if not exists idx_restaurants_price on public.restaurants (price);
create index if not exists idx_restaurants_rating on public.restaurants (rating);
create index if not exists idx_restaurants_created_at on public.restaurants (created_at desc);

-- =========================================================================
-- 3. Row Level Security (RLS) 활성화 (데이터 접근 보안)
-- =========================================================================
alter table public.restaurants enable row level security;

-- =========================================================================
-- 4. 인가된 사용자(authenticated) 대상 읽기, 쓰기, 수정 정책 부여
-- =========================================================================
-- 4-1. 인가된 사용자 누구나 맛집 데이터 조회(SELECT) 가능
drop policy if exists "Allow authenticated users to read restaurants" on public.restaurants;
create policy "Allow authenticated users to read restaurants"
on public.restaurants for select
to authenticated
using (true);

-- 4-2. 인가된 사용자 누구나 CSV 또는 신규 맛집 데이터 누적 적재(INSERT/UPSERT) 가능
drop policy if exists "Allow authenticated users to insert restaurants" on public.restaurants;
create policy "Allow authenticated users to insert restaurants"
on public.restaurants for insert
to authenticated
with check (true);

-- 4-3. 인가된 사용자의 맛집 정보 및 추천수/리뷰 업데이트(UPDATE) 허용
drop policy if exists "Allow authenticated users to update restaurants" on public.restaurants;
create policy "Allow authenticated users to update restaurants"
on public.restaurants for update
to authenticated
using (true);

-- =========================================================================
-- 5. Realtime 복제 활성화 (새 식당 등록 및 CSV 누적 시 실시간 전파)
-- =========================================================================
alter publication supabase_realtime add table public.restaurants;
